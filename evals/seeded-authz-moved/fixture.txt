const express = require('express')
const { requireAuth, requireRole } = require('./middleware/auth')
const adminRouter = require('./routes/admin')
const reportsRouter = require('./routes/reports')

const app = express()

// Admin routes: every handler under /admin is for staff with the admin role.
app.use('/admin', requireAuth, requireRole('admin'), adminRouter)
// Reports: any signed-in seller sees their own reports.
app.use('/reports', requireAuth, reportsRouter)

module.exports = app
@@@ base src/middleware/auth.js
const jwt = require('jsonwebtoken')

function requireAuth(req, res, next) {
  try {
    req.user = jwt.verify((req.headers.authorization || '').replace(/^Bearer /, ''), process.env.JWT_SECRET)
    next()
  } catch {
    res.status(401).end()
  }
}

const requireRole = (role) => (req, res, next) => (req.user.roles.includes(role) ? next() : res.status(403).end())

module.exports = { requireAuth, requireRole }
@@@ base src/routes/admin.js
const express = require('express')
const { db } = require('../db')
const { toCsv } = require('../lib/csv')

const router = express.Router()

router.get('/sellers', async (req, res) => {
  res.json(await db('sellers').select('id', 'name', 'status'))
})

router.get('/payouts/export', async (req, res) => {
  const rows = await db('payouts').join('sellers', 'sellers.id', 'payouts.seller_id')
    .select('payouts.id', 'sellers.name', 'sellers.bank_account', 'sellers.sort_code', 'payouts.amount')
  res.type('text/csv').send(toCsv(rows))
})

module.exports = router
