@@@ base src/lib/collections.js
// The item with the given id, or undefined.
function findById(list, id) {
  return list.find((item) => item.id === id)
}

// A Map from id to item, for repeated lookups.
function indexById(list) {
  return new Map(list.map((item) => [item.id, item]))
}

module.exports = { findById, indexById }
@@@ base src/reports/order-summaries.js
const { indexById } = require('../lib/collections')

// Nightly export: every order of the day with its customer's name. Runs over the whole
// table, about 200,000 orders and 60,000 customers.
function buildOrderSummaries(orders, customers) {
  const customersById = indexById(customers)
  return orders.map((order) => ({
    id: order.id,
    total: order.total,
    customer: customersById.get(order.customerId).name,
  }))
}

module.exports = { buildOrderSummaries }
@@@ head src/reports/order-summaries.js
const { findById } = require('../lib/collections')

// Nightly export: every order of the day with its customer's name. Runs over the whole
// table, about 200,000 orders and 60,000 customers.
function buildOrderSummaries(orders, customers) {
  return orders.map((order) => ({
    id: order.id,
    total: order.total,
    // A deleted customer has no row any more: show "unknown" rather than crash.
    customer: findById(customers, order.customerId)?.name ?? 'unknown',
  }))
}

module.exports = { buildOrderSummaries }
@@@ head test/order-summaries.test.js
const { buildOrderSummaries } = require('../src/reports/order-summaries')

  expect(rows[0].customer).toBe('unknown')
})
