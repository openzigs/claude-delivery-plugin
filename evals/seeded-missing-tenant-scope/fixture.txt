@@@ head src/invoices/repository.ts
import { db } from '../db'

export async function listInvoices(tenantId: string) {
  return db('invoices').where({ tenant_id: tenantId })
}

// Called by GET /tenants/:tenantId/invoices/:invoiceId
export async function getInvoice(tenantId: string, invoiceId: string) {
  return db('invoices').where({ id: invoiceId }).first()
}

export async function markPaid(tenantId: string, invoiceId: string) {
  return db('invoices').where({ tenant_id: tenantId, id: invoiceId }).update({ paid: true })
}
@@@ head src/invoices/routes.ts
import { Router } from 'express'
import { listInvoices, getInvoice, markPaid } from './repository'

export const router = Router()

router.get('/tenants/:tenantId/invoices', async (req, res) => {
  res.json(await listInvoices(req.params.tenantId))
})

router.get('/tenants/:tenantId/invoices/:invoiceId', async (req, res) => {
  const invoice = await getInvoice(req.params.tenantId, req.params.invoiceId)
  if (!invoice) return res.status(404).end()
  res.json(invoice)
})

router.post('/tenants/:tenantId/invoices/:invoiceId/paid', async (req, res) => {
  await markPaid(req.params.tenantId, req.params.invoiceId)
  res.status(204).end()
})
