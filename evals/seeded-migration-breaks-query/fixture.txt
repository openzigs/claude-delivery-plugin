@@@ base src/orders/repository.js
const { query } = require('../db')

async function createOrder(customerId, total) {
  const [row] = await query('INSERT INTO orders (customer_id, total) VALUES ($1, $2) RETURNING id', [customerId, total])
  return row.id
}

async function orderTotal(id) {
  const [row] = await query('SELECT total FROM orders WHERE id = $1', [id])
  return row ? row.total : null
}

module.exports = { createOrder, orderTotal }
@@@ base src/reports/revenue.js
const { query } = require('../db')

// Daily revenue for the finance dashboard, run every night by the scheduler.
async function dailyRevenue(since) {
  return query(
    "SELECT date_trunc('day', created_at) AS day, SUM(total) AS revenue FROM orders WHERE created_at >= $1 GROUP BY 1 ORDER BY 1",
    [since],
  )
}

module.exports = { dailyRevenue }
