import { loadStripe } from '@stripe/stripe-js'

export async function startCheckout(clientSecret) {
  const stripe = await loadStripe(window.__CONFIG__.stripePublishableKey)
  return stripe.confirmPayment({ clientSecret, confirmParams: { return_url: `${location.origin}/thanks` } })
}
