# inventory

Stock levels for the warehouse service.

## Install

Requires Node 20.

```
npm install
```

## Running the tests

Run `npm test` for the unit suite. The integration suite needs a local database:

    docker compose up -d db
    npm run test:integration

## License

MIT

// Integration tests: they need the database from docker-compose.yml (postgres on localhost:5432,
// password "postgres"; test/integration/setup.js reads DATABASE_URL or falls back to that).
module.exports = {
  testMatch: ['<rootDir>/test/integration/**/*.test.js'],
  setupFiles: ['<rootDir>/test/integration/setup.js'],
}
@@@ base test/integration/setup.js
process.env.DATABASE_URL ||= 'postgres://postgres:postgres@localhost:5432/postgres'
@@@ base test/integration/stock.test.js
const { Client } = require('pg')

test('the stock table exists', async () => {
  const db = new Client({ connectionString: process.env.DATABASE_URL })
  await db.connect()
  const { rows } = await db.query("SELECT to_regclass('public.stock') AS t")
  await db.end()
  expect(rows[0].t).toBe('stock')
})
@@@ base docker-compose.yml
services:
  db:
    image: postgres:16
    ports: ["5432:5432"]
    environment:
      POSTGRES_PASSWORD: postgres
