@@@ head src/users/dedupe.js
// Find duplicate sign-ups: two accounts with the same email are one person.
// Returns [original, duplicate] pairs in sign-up order.
function findDuplicates(users) {
  const seen = new Map()
  const pairs = []
  for (const user of users) {
    if (seen.has(user.email)) pairs.push([seen.get(user.email), user])
    else seen.set(user.email, user)
  }
  return pairs
}

module.exports = { findDuplicates }
@@@ head test/dedupe.test.js
const users = require('./fixtures/users.json')
const { findDuplicates } = require('../src/users/dedupe')

test('finds every duplicate sign-up', () => {
  const pairs = findDuplicates(users)
  expect(pairs.map(([a, b]) => [a.id, b.id])).toEqual([[1, 3], [2, 5]])
})

test('finds no duplicates in a list of distinct users', () => {
  expect(findDuplicates(users.filter((u) => u.id !== 3 && u.id !== 5))).toEqual([])
})
