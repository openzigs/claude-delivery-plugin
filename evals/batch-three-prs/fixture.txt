const router = require('express').Router()
const { query } = require('../db')
const { slugify } = require('../lib/slug')

// Every note belongs to one user; a user only ever sees or changes their own notes.
router.get('/notes', async (req, res) => {
  res.json(await query('SELECT * FROM notes WHERE user_id = $1 ORDER BY created_at DESC', [req.user.id]))
})

router.get('/notes/:id', async (req, res) => {
  const [note] = await query('SELECT * FROM notes WHERE id = $1 AND user_id = $2', [req.params.id, req.user.id])
  if (!note) return res.status(404).end()
  res.json(note)
})

router.post('/notes', async (req, res) => {
  const { title, body, dueAt = null } = req.body
  const [note] = await query(
    'INSERT INTO notes (user_id, title, slug, body, due_at) VALUES ($1, $2, $3, $4, $5) RETURNING *',
    [req.user.id, title, slugify(title), body, dueAt],
  )
  res.status(201).json(note)
})

module.exports = router

// Notes whose due date has passed, for the hourly reminder email.
function overdue(notes, now = new Date()) {
  return notes.filter((note) => note.due_at !== null && note.due_at < now)
}

// URL slug for a note title: lower case, accents removed, runs of other characters collapsed
// to one hyphen, at most 80 characters, and never empty.
function slugify(title) {
  const plain = String(title).toLowerCase().normalize('NFKD').replace(/[̀-ͯ]/g, '')
  return plain.replace(/[^a-z0-9]+/g, '-').replace(/^-+|-+$/g, '').slice(0, 80) || 'note'
}

module.exports = { slugify }
